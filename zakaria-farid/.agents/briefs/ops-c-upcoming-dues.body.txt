export interface UpcomingDue {
  id: string;                 // IN: `sch-${schedule.schedule_id}`; OUT: `pay-${cost.item_id}-${inst.installment_id}`

  title: string;
  party: string;
  amount: Decimal;            // remaining = nominal - paid (> 0)
  schedule?: ERPInstallmentSchedule;
  contract?: ERPContract;
  cost?: ERPPropertyCostItem;
  installment?: ERPPayableInstallment;
}
export interface UpcomingDuesResult {
  items: UpcomingDue[];       // ALL matching items, sorted dueDate asc, tie -> id asc
  inTotal: Decimal; outTotal: Decimal; inCount: number; outCount: number; overdueCount: number;
  horizonEnd: string;         // todayStr + horizonDays (toLocalDateStr)
}

export function buildUpcomingDues(params: {
  schedules?: ERPInstallmentSchedule[] | null;
  contracts?: ERPContract[] | null;
  propertyCosts?: ERPPropertyCostItem[] | null;
  todayStr: string;
  horizonDays?: number;       // default 30
  isAr: boolean;
}): UpcomingDuesResult

Rules:
- horizonEnd = date of (todayStr + horizonDays) computed like getTreasuryPeriodRange does (new Date(y, m-1, d+N)).
- Include an item only if dueDate <= horizonEnd (overdue items always qualify) and remaining > 0.
- IN from schedules: status in ['Pending','Partially Paid','Defaulted'] only; contract = contracts.find(c =>
  c.contract_id === s.contract_id); skip if no contract or contract.status !== 'Active'.
  remaining = D(s.nominal_value || 0).minus(D(s.amount_paid || 0)).
  title (isAr): s.tranche_number === 0 ? `دفعة المقدم — ${contract.contract_number}` : `القسط ${s.tranche_number} — ${contract.contract_number}`
  title (en):   tranche 0 ? `Down payment — ${n}` : `Installment ${t} — ${n}`
  party = contract.buyer_name || '—'
- OUT from propertyCosts[].payable_installments: skip inst.status === 'PAID';
  remaining = D(inst.amount_egp || 0).minus(D(inst.paid_amount_egp || 0)).
  title = isAr ? (inst.title_ar || cost.item_name_ar) : (inst.title_en || inst.title_ar || cost.item_name_en || cost.item_name_ar)
  party = cost.supplier_contractor || '—'
- Never mutate inputs.
