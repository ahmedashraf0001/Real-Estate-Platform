export type CashAccountCode = '101000' | '102000';
export type TreasuryPeriod = 'today' | '7d' | 'month' | 'all';
export type CashMovementKind =
  | 'customer_collection' | 'partner_injection' | 'loan_inflow' | 'other_inflow'
  | 'contractor_payment' | 'site_expense' | 'operating_expense' | 'tax_remittance'
  | 'customer_refund' | 'partner_distribution' | 'loan_repayment' | 'other_outflow'
  | 'internal_transfer';
export type CashDirection = 'IN' | 'OUT' | 'TRANSFER';

export interface TreasuryMovement {
  id: string;                 // entry.entry_id
  entryNumber: string;        // entry.entry_number || entry.entry_id
  date: string;               // entry.entry_date (YYYY-MM-DD); fallback toLocalDateStr(new Date(entry.created_at))
  time: string;               // local HH:mm from entry.created_at; '' if created_at missing/invalid
  timestamp: number;          // Date.parse(entry.created_at); if NaN -> Date.parse(date + 'T00:00:00'); if NaN -> 0
  kind: CashMovementKind;
  direction: CashDirection;
  account: CashAccountCode;           // IN/OUT: the cash account moved. TRANSFER: the DEBITED cash account (destination)
  fromAccount?: CashAccountCode;      // TRANSFER only: the CREDITED cash account (source)
  amount: Decimal;            // always positive
  signedAmount: Decimal;      // IN: +amount, OUT: -amount, TRANSFER: D(0)
  opposingAccountCode: string; // see rule 3; '' for TRANSFER
  originalDescription: string; // entry.description || ''
  description: string;        // see rule 5
  counterparty: string;       // see rule 6
  contract?: ERPContract;
  entry: ERPJournalEntry;
  balanceAfter: Decimal;      // total cash (101000+102000) after this movement, chronological (rule 7)
}

export interface TreasuryAccountSummary {
  code: CashAccountCode | 'TOTAL';
  opening: Decimal; inflow: Decimal; outflow: Decimal; closing: Decimal;
}
export interface TreasuryKindSummary {
  kind: CashMovementKind; direction: CashDirection; count: number; amount: Decimal;
  share: number | null;       // amount / total of same direction * 100, rounded to 1 decimal; null for TRANSFER or when direction total is 0
}
export interface TreasurySummary {
  period: TreasuryPeriod; from: string | null; to: string | null;
  accounts: TreasuryAccountSummary[];   // exactly 3 rows in order: '101000', '102000', 'TOTAL'
  inflow: Decimal; outflow: Decimal; net: Decimal; count: number;  // company level, TRANSFER excluded from inflow/outflow, included in count
  byKind: TreasuryKindSummary[];        // only kinds with count > 0; order: IN rows, then OUT rows, then TRANSFER; inside each group amount desc, tie -> kind asc
}

3. Rules for buildTreasuryMovements(entries: ERPJournalEntry[] | null | undefined,
   contracts: ERPContract[] | null | undefined, isAr: boolean): TreasuryMovement[]

 Per entry (skip null entries, treat missing lines as []):
 a. cashDr[code] = sum debit_amount, cashCr[code] = sum credit_amount of lines with account_code
    '101000' / '102000' (use D(x || 0)). netByCode = Dr - Cr per cash code.
 b. If both cash codes have non-zero net AND one is positive and the other negative
    -> TRANSFER: account = code with positive net, fromAccount = code with negative net,
    amount = positive net of destination, kind 'internal_transfer', opposingAccountCode ''.
 c. Else totalNet = net101000 + net102000. If totalNet == 0 -> skip the entry entirely.
    If totalNet > 0 -> direction IN; if < 0 -> OUT. amount = |totalNet|.
    account = the cash code with the largest |net| (tie -> '101000').
 d. opposing line = among NON-cash lines, the one with the largest max(debit, credit);
    tie -> lowest line_number; if line_number missing keep array order. opposingAccountCode =
    its account_code or '' if none.
 e. kind (use first matching rule, prefix match on opposingAccountCode):
    IN:  '301','302' -> partner_injection; '202' -> loan_inflow;
         '103','104','203','4' -> customer_collection; else other_inflow.
    OUT: '201' -> contractor_payment; '105','15','603' -> site_expense;
         '204','604' -> tax_remittance; '206' -> customer_refund; '303' -> partner_distribution;
         '202' -> loan_repayment; '5','6' -> operating_expense; else other_outflow.
    Order matters: check '603' before '6', '604' before '6'.

4. Contract match (first hit wins): any line.contract_id equal to contract.contract_id;
   else entry.source_entity_id === contract.contract_id; else contract.contract_number is a
   non-empty substring of entry.description.

5. description:
   const hasArabic = /[؀-ۿ]/.test(originalDescription)
   if (!isAr) -> originalDescription || label.en
   if (isAr && hasArabic) -> originalDescription
   if (isAr && !hasArabic) -> label.ar + (contract ? ` — عقد ${contract.contract_number}` : '')
   (label = CASH_MOVEMENT_KIND_LABELS[kind]). The UI shows originalDescription as a secondary line
   when it differs from description — not your concern here.

6. counterparty (isAr ? ar : en):
   contract -> contract.buyer_name
   partner_injection | partner_distribution -> 'الشركاء' | 'Partners'
   tax_remittance -> 'مصلحة الضرائب' | 'Tax authority'
   internal_transfer -> 'داخلي' | 'Internal'
   loan_inflow | loan_repayment -> 'البنك' | 'Bank'
   anything else -> '—'

7. Ordering + running balance: sort ascending by (timestamp, then date string, then id).
   Walk in that order with running = D(0); for non-TRANSFER add signedAmount; TRANSFER adds 0.
   balanceAfter = running after the row. Return the array sorted DESCENDING (newest first)
   — i.e. reverse of the ascending walk.

8. export function toLocalDateStr(d: Date): string  -> `${YYYY}-${MM}-${DD}` from local getters.

9. export function getTreasuryPeriodRange(period: TreasuryPeriod, todayStr: string):
   { from: string | null; to: string | null }
   today -> {from: todayStr, to: todayStr}
   7d    -> {from: todayStr minus 6 days, to: todayStr}  (compute with new Date(y, m-1, d-6) + toLocalDateStr)
   month -> {from: `${todayStr.slice(0,7)}-01`, to: todayStr}
   all   -> {from: null, to: null}

10. export function isInTreasuryPeriod(m: TreasuryMovement, range: {from: string|null; to: string|null}): boolean
   -> (!from || m.date >= from) && (!to || m.date <= to)

11. export function summarizeTreasury(movements: TreasuryMovement[], period: TreasuryPeriod,
   todayStr: string): TreasurySummary
   range = getTreasuryPeriodRange(period, todayStr).
   For each cash code C:
     closing = sum over ALL movements with (!range.to || date <= range.to) of effect on C, where
       effect on C: IN/OUT with account C -> signedAmount; TRANSFER: +amount if account == C,
       -amount if fromAccount == C; else 0.
     inflow  = sum over in-period movements of positive effects on C (IN to C, TRANSFER into C)
     outflow = sum over in-period movements of |negative effects| on C (OUT from C, TRANSFER out of C)
     opening = closing - inflow + outflow
   TOTAL row = per-field sum of the two code rows EXCEPT inflow/outflow of TOTAL exclude TRANSFER
   (so TOTAL.opening = sum openings, TOTAL.closing = sum closings, TOTAL.inflow = company inflow,
   TOTAL.outflow = company outflow). Check TOTAL.opening + inflow - outflow == closing holds.
   inflow/outflow/net/count = company level over in-period movements (TRANSFER excluded from
   inflow/outflow; counted in count). byKind per rule 1.

12. Never mutate inputs. No console logging. No `any` except where unavoidable on line fields
   (none expected: ERPJournalLine has account_code, debit_amount, credit_amount, contract_id?, line_number).
   Verify field names in src/lib/erp/types.ts before use; if a name differs, use the real one and
   list it in the report.
